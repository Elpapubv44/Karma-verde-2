```typescript
import { useState, useEffect } from "react";
import { PaperCard, PaperButton } from "@/components/paper/Paper";
import { Sparkles, Globe, ExternalLink, Leaf, Lightbulb, Check } from "lucide-react";
import { generateLocalAnalysis } from "@/lib/gemini-qr-analyzer";

export interface QrIntelligenceResult {
  objeto: string;
  origen: string;
  tipoMaterial: string;
  reciclabilidad: string;
  descripcion: string;
  impactoEstimado: string;
  consejo: string;
}

interface SmartQrModalProps {
  code: string | null;
  onClose: () => void;
  onConfirmCanje?: () => void;
}

export function SmartQrModal({ code, onClose, onConfirmCanje }: SmartQrModalProps) {
  const [data, setData] = useState<QrIntelligenceResult | null>(null);
  const [isAiPowered, setIsAiPowered] = useState(false);

  useEffect(() => {
    if (!code) {
      setData(null);
      return;
    }

    let isMounted = true;
    const localFallback = generateLocalAnalysis(code);
    setData(localFallback);
    setIsAiPowered(false);

    async function analyze() {
      try {
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 3500);

        const res = await fetch("/api/analyze-qr", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ code }),
          signal: controller.signal,
        });
        clearTimeout(timeoutId);

        if (res.ok) {
          const json = await res.json();
          if (isMounted && json.analysis && Object.keys(json.analysis).length > 0) {
            setData(json.analysis);
            setIsAiPowered(json.source === "gemini");
          }
        }
      } catch {
        // Fallback silencioso y fluido
      }
    }

    analyze();
    return () => { isMounted = false; };
  }, [code]);

  if (!code || !data) return null;
  const isUrl = code.startsWith("http://") || code.startsWith("https://");

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-ink/65 p-4 backdrop-blur-xs">
      <PaperCard className="relative w-full max-w-lg p-6 space-y-4 border-2 border-primary/40 bg-cream">
        <div className="flex items-start justify-between gap-2 border-b border-kraft/60 pb-3">
          <div className="flex items-center gap-2">
            <div className="grid h-10 w-10 place-items-center rounded-2xl bg-primary text-primary-foreground">
              <Sparkles className="h-5 w-5" />
            </div>
            <div>
              <h3 className="text-xl font-bold text-ink">Investigador Inteligente de QR</h3>
              <p className="text-xs text-muted-foreground">Decodificación y análisis en tiempo real</p>
            </div>
          </div>
          <button onClick={onClose} className="p-1 text-muted-foreground hover:text-ink">✕</button>
        </div>

        <div className="space-y-3">
          <div className="rounded-xl border border-kraft/70 bg-white/70 p-3 space-y-1">
            <span className="text-[10px] font-black uppercase text-earth">Objeto / Contenido Identificado</span>
            <p className="text-sm font-extrabold text-ink">{data.objeto}</p>
          </div>

          <div className="grid grid-cols-2 gap-2 text-xs">
            <div className="rounded-xl border border-kraft/70 bg-white/70 p-2.5">
              <p className="text-[10px] font-bold text-muted-foreground uppercase">Origen</p>
              <p className="font-bold text-ink">{data.origen}</p>
            </div>
            <div className="rounded-xl border border-kraft/70 bg-white/70 p-2.5">
              <p className="text-[10px] font-bold text-muted-foreground uppercase">Material</p>
              <p className="font-bold text-leaf">{data.tipoMaterial}</p>
            </div>
          </div>

          <div className="rounded-xl border border-leaf/40 bg-leaf/10 p-3 text-xs text-ink/90">
            <p className="font-semibold">{data.descripcion}</p>
            <p className="mt-2 text-leaf font-bold">🌱 {data.impactoEstimado}</p>
          </div>
        </div>

        <div className="flex gap-2 pt-2">
          {isUrl && (
            <a href={code} target="_blank" rel="noreferrer" className="flex-1">
              <PaperButton variant="primary" className="w-full justify-center">
                <Globe className="h-4 w-4 mr-1" /> Visitar Enlace
              </PaperButton>
            </a>
          )}
          <PaperButton variant="ghost" onClick={onClose} className="flex-1 justify-center">
            Cerrar
          </PaperButton>
        </div>
      </PaperCard>
    </div>
  );
}
```
